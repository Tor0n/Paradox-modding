#Eldar
country_event = {
	id = WrWrld_game_start_aeldari.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = Eldar_gos
	}
	immediate = {
		save_global_event_target_as = Eldar_evnt_trgt_country
		owner = { country_event = { id = WrWrld_game_start_aeldari.101 days = 1 } }
		communication_with_all = yes
		add_resource = { energy = 25000	}
		add_resource = { minerals = 14000 }
		add_resource = { food = 14000 }
		add_resource = { consumer_goods = 9500 }
		add_resource = { alloys = 17500	}
		add_resource = { volatile_motes = 1000 }
		add_resource = { rare_crystals = 400 }
		add_resource = { exotic_gases = 400 }
		
		give_technology_eldar = yes
		add_tradition_eldar = yes
		
		every_system = {
			limit = { has_star_flag = Eldar_gos }
			if = {
				limit = {
					OR = {
						has_star_flag = Eldar_Habitat
						has_star_flag = Eldar_gaia
					}
				}
				create_starbase = {
					owner = ROOT
					size = starbase_starfortress
					module = shipyard
					module = shipyard
				}
			}
			else = {
				#limit = { has_star_flag = Eldar_empty }
				create_starbase = {
					owner = ROOT
					size = starbase_outpost
				}
			}
			every_system_planet = {
				if = {
					limit = { is_colonizable = no }
					create_deposits_stations = yes
				}
				if = {
					limit = { is_colonizable = yes }
					if = {
						limit = {
							has_planet_flag = Eldar_world
							NOT = { has_planet_flag = habitat }
						}
						create_colony = { owner = ROOT }
						while = {
							count = 5
							add_district_and_planet_size_if_needed_effect = { district = district_city }
						}
						while = {
							count = 5
							add_district_and_planet_size_if_needed_effect = { district = district_industrial }
						}
						while = {
							count = 1
							add_district_and_planet_size_if_needed_effect = { district = district_generator }
						}
						while = {
							count = 6
							add_district_and_planet_size_if_needed_effect = { district = district_mining }
						}
						while = {
							count = 2
							add_district_and_planet_size_if_needed_effect = { district = district_farming }
						}
						while = {
							count = 30
							create_pop_group = { species = owner_main_species }
						}
						add_building = building_system_capital
						add_building = building_planetary_shield_generator
						add_building = building_fortress
						#add_building = building_class_4_singularity
						#add_building = building_archaeo_refinery
						#add_building = building_production_center
						add_building = building_ministry_production
						add_building = building_foundry_1
						add_building = building_mineral_purification_hub
					}
					else_if = {
						limit = { has_planet_flag = habitat }
						create_colony = { owner = ROOT }
						while = {
							count = 3
							add_district_and_planet_size_if_needed_effect = { district = district_hab_housing }
						}
						while = {
							count = 4
							add_district_and_planet_size_if_needed_effect = { district = district_hab_industrial }
						}
						while = {
							count = 33
							create_pop_group = { species = owner_main_species }
						}
						add_building = building_hab_system_capital
						add_building = building_planetary_shield_generator
						add_building = building_fortress
						#add_building = building_class_4_singularity
						#add_building = building_refinery
						add_building = building_chemical_plant
						add_building = building_ministry_production
						add_building = building_bio_reactor_2
						add_building = building_hydroponics_farm
						add_building = building_hydroponics_farm
						add_building = building_foundry_1
					}
				}
				
			}
			#*-/ врата \-*#
			if = {
				limit = {
					OR = {
						has_star_flag = Eldar_Habitat
						has_star_flag = Eldar_gaia
					}
				}
				spawn_megastructure = {
					type = gateway_final
					graphical_culture = ROOT
					orbit_distance = 265
				}
			}
		}
	}
}
