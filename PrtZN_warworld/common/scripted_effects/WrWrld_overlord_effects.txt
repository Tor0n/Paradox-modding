add_agrarian_leader_trait = {
	random_list = {
		1 = { add_trait_no_notify = leader_trait_agrarian_agronome }
		1 = { add_trait_no_notify = leader_trait_agrarian_planter }
	}
}

remove_agrarian_leader_trait = {
	if = {
		limit = { has_trait = leader_trait_agrarian_agronome }
		remove_trait = leader_trait_agrarian_agronome
	}
	else_if = {
		limit = { has_trait = leader_trait_agrarian_planter }
		remove_trait = leader_trait_agrarian_planter
	}
}
